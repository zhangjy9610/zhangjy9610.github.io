\begin{problem}[习题3.1.1]
求解如下具有Neumann边界条件的波动方程，其中常数 $c>0$
\[
\begin{cases}
u_{tt}-c^2u_{xx}=0&t>0,~0<x<\pi,\\
u(0,x)=\cos x,~~u_t(0,x)=0& 0\leqslant x\leqslant \pi,\\
u_x(t,0)=u_x(t,\pi)=0& t\geqslant 0.
\end{cases}
\]
\end{problem}

\begin{problem}[习题3.1.5]
设$A,B$是常数，求解如下方程
\[
\begin{cases}
u_{tt}-u_{xx}=0&~~~t>0,~0<x<\pi\\
u(0,x)=0,~u_t(0,x)=0&~~~0\leqslant x\leqslant \pi\\
u_x(t,0)=At,~~u_x(t,\pi)=Bt&~~~t\geqslant 0.
\end{cases}
\]
\end{problem}


\begin{problem}[2025期末第三题，习题3.1.6改编]
 用分离变量法求解方程，其中$\omega>0$是常数。
    \[
    \begin{cases}
        u_{tt}-u_{xx}=2\sin 2x\cos x \cos(\omega t) &t>0,~x\in(0,\pi),\\
        u(t,0)=u(t,\pi)=0&t\geqslant 0,\\
         u(0,x)=u_t(0,x)=0&x\in [0,\pi].
    \end{cases}
    \]然后讨论 $\omega$ 为何值时方程的解一致有界？即$\sup\limits_{t>0,x\in(0,\pi)}|u(t,x)|< \infty$.
\end{problem}


\begin{problem}[2025年小测第三题]
考虑一根两端固定的弦发生阻尼振动的方程，阻尼系数$d\in(0,2]$.
\[
\begin{cases}
u_{tt}-u_{xx}+du_t=0&~~~t>0,~0<x<\pi;\\
u(t,0)=u(t,\pi)=0&~~~t\geqslant 0;\\
u(0,x)=\vp(x),\q u_t(0,x)=0&~~~0\leqslant x\leqslant\pi.
\end{cases}
\]其中初值 $\vp\in C^3([0,\pi])$ 满足相容性条件 $\vp(0)=\vp(\pi)=\vp''(0)=\vp''(\pi)=0$.
\begin{itemize}
    \item [(1)]  用分离变量法求出方程解的表达式（用含 $\vp$ 的式子表示）。
    \item [(2)]  若 $0<d<2$，证明：存在常数 $C>0$，使得(1)中求得的解满足 $|u(t,x)|\leqslant Ce^{-\frac{d}{2}t}$ 对任意 $t>0,x\in(0,\pi)$ 都成立，其中常数 $C$ 仅依赖初值 $\vp$.
    \item [(3)] 若 $d=2$，是否存在不恒为零且满足相容性条件的 $\vp(x)$，使得对任意固定的 $x\in(0,\pi)$ 有 $\limsup\limits_{t\to+\infty} e^{t}|u(t,x)|=+\infty$ 成立？若是，请构造一个 $\vp(x)$，否则证明这个式子不对。
\end{itemize}
\end{problem}


\begin{problem}[习题3.2.3]
考虑具有Neumann边界条件的热传导方程，其中$k\geq0$是常数
\[
\begin{cases}
u_{t}-u_{xx}=k(1-u)&~~~t>0,0<x<\pi\\
u(0,x)=\varphi(x)&~~~0\leqslant x\leqslant \pi\\
u_x(t,0)=u_x(t,\pi)=0&~~~t\geqslant 0.
\end{cases}
\]求解这个方程，并计算$\lim\limits_{t\to+\infty} u(t,x)$. （提示：$k=0$和$k>0$答案不同。）
\end{problem}


\begin{problem}[习题3.2.5]
考虑如下热传导方程的初边值问题
\[
u_t-u_{xx}=0~t>0,x\in(0,\pi);\q u(0,x)=\vp(x)\in C^2([0,\pi])~0\leqslant x\leqslant \pi;\q u(t,0)=u(t,\pi)=0,~t\geqslant 0.
\]
\begin{itemize}
\item [(1)] 证明：方程的解满足估计$\int_0^\pi u(t,x)^2\dx\leqslant e^{-2t}\int_0^\pi \vp(x)^2\dx.$
\item [(2)] 证明：存在常数$C>0$使得$|u(t,x)|\leqslant Ce^{-t}$对任意$t>0,x\in[0,\pi]$成立。
\end{itemize}
提示：无论使用能量法还是分离变量法，可能都需要使用Parseval恒等式；对(2)，思考为什么这里假设了$\vp\in C^2([0,\pi])$，具有该光滑性的函数的傅立叶系数具有怎样的阶？
\end{problem}

\begin{problem}[补充题，Dirichlet-to-Neumann 算子]
设$B\subset\R^2$是单位开圆盘，$S=\p B$是单位圆周，$h\in C^\infty(S)$ 是给定的实值函数，其在单位圆盘内的调和延拓记作 $u$，即 $\lap u=0$ in $B$，$u|_{\p B}=h$. 设 $\R^2$ 上的极坐标为 $(r,\theta)$，今定义\textbf{调和函数的 Dirichlet-to-Neumann (DtN) 算子} $\NH$ 如下： $$\NH h:=\dfrac{\p u}{\p N}\bigg|_{S}=\p_r u|_{r=1}.$$ 
证明如下结论
\begin{itemize}
\item [(1)] $\NH^2h=-\p_\theta^2h$.
\item [(2)]  $\NH$ 是非负对称算子，即对任意 $g,h\in C^\infty(S)$成立 $$\int_0^{2\pi} h\,(\NH g)\,\mathrm{d}\theta = \int_0^{2\pi} (\NH h)\, g\,\mathrm{d}\theta,\qquad \int_0^{2\pi} h\,(\NH h)\,\mathrm{d}\theta\geqslant 0.$$ 并计算 $\ker \NH$.
\item [(3)] 给定 $f\in C^\infty(S)$，则方程 $\NH h = f$ 存在光滑解当且仅当 $\int_0^{2\pi}f(\theta)\,\mathrm{d}\theta = 0$.
\end{itemize}
\end{problem}

\begin{problem}[习题3.3.6，选做] 
设$0<\omega<2\pi$，定义扇形区域 $W_\omega=\{(r,\theta):0<r<1,\ 0<\theta<\omega\}.$ 考虑函数
\[
u(r,\theta)=r^{\frac{\pi}{\omega}}\sin\frac{\pi\theta}{\omega}.
\]
\begin{itemize}
\item[(1)] 证明$u$在$W_\omega$中调和，并且在两条直边$\theta=0$与$\theta=\omega$上连续地取零值。

\item[(2)] 证明 $|\nab u(r,\theta)|=\dfrac{\pi}{\omega}r^{\frac{\pi}{\omega}-1}$.

\item[(3)] 分别讨论$\omega<\pi$、$\omega=\pi$与$\omega>\pi$时，$|\nab u|$在顶点$r=0$附近的行为，并说明这一例子为何表明凹角可能破坏调和函数到边界的一阶连续可微性。
\end{itemize}
\end{problem}
